## Usage

Each section arrives as a slot, so the panel only arranges them under the Account title.

```tsx
import { useRef } from 'react';
import { UserProfileProfilePanelView } from '@clerk/mosaic/features/user-profile/user-profile-profile-panel.view';
import { UserProfileProfileSection } from '@clerk/mosaic/features/user-profile/user-profile-profile-section/user-profile-profile-section';
import { UserProfileEmailSection } from '@clerk/mosaic/features/user-profile/user-profile-email-section/user-profile-email-section';
import { UserProfilePhoneSection } from '@clerk/mosaic/features/user-profile/user-profile-phone-section/user-profile-phone-section';
import { UserProfileConnectedAccountsSection } from '@clerk/mosaic/features/user-profile/user-profile-connected-accounts-section/user-profile-connected-accounts-section';

const titleRef = useRef<HTMLDivElement>(null);

<UserProfileProfilePanelView
  titleRef={titleRef}
  profileSlot={<UserProfileProfileSection />}
  emailSlot={<UserProfileEmailSection />}
  phoneSlot={<UserProfilePhoneSection />}
  connectedAccountsSlot={<UserProfileConnectedAccountsSection fallbackFocus={() => titleRef.current} />}
/>
```
